cmake_minimum_required(VERSION 3.21)
project(bgfx_fetchcontent_parent LANGUAGES C CXX)

if(NOT BGFX_WRAPPER_SOURCE_DIR)
	message(FATAL_ERROR "Set BGFX_WRAPPER_SOURCE_DIR to the bgfx.cmake checkout")
endif()
file(TO_CMAKE_PATH "${BGFX_WRAPPER_SOURCE_DIR}" BGFX_WRAPPER_SOURCE_DIR)

include(FetchContent)
set(BGFX_BUILD_EXAMPLES OFF CACHE BOOL "")
set(BGFX_BUILD_TOOLS OFF CACHE BOOL "")
set(BGFX_INSTALL OFF CACHE BOOL "")
if(BGFX_PARENT_USE_ADD_SUBDIRECTORY)
	if(NOT BX_DIR OR NOT BGFX_DIR)
		message(FATAL_ERROR "The add_subdirectory case requires explicit BX_DIR and BGFX_DIR overrides")
	endif()
	add_subdirectory("${BGFX_WRAPPER_SOURCE_DIR}" wrapper)
else()
	FetchContent_Declare(bgfx SOURCE_DIR "${BGFX_WRAPPER_SOURCE_DIR}")
	FetchContent_MakeAvailable(bgfx)

	FetchContent_GetProperties(bgfx_source SOURCE_DIR upstream_bgfx_dir)
	if(NOT EXISTS "${upstream_bgfx_dir}/include/bgfx/bgfx.h")
		message(FATAL_ERROR "The upstream bgfx sources were not populated separately from the wrapper")
	endif()
endif()
if(NOT TARGET bgfx)
	message(FATAL_ERROR "The wrapper did not create the bgfx library target")
endif()
